(function () {
  var D = JSON.parse(document.getElementById('run-data').textContent);
  var M = D.meta, steps = D.steps, phases = D.phases, key = (D.key && D.key.rows) || [], acMap = D.ac || {};
  var KEY = 'acceptance-run:' + M.storageKey;
  var state = { data: {}, res: {}, note: {}, cur: 0, view: D.domain ? 'domain' : 'run' };
  try { var raw = localStorage.getItem(KEY); if (raw) state = Object.assign(state, JSON.parse(raw)); } catch (e) {}
  function save() { try { localStorage.setItem(KEY, JSON.stringify(state)); } catch (e) {} }
  // Automated results: a newer run replaces what the viewer marked by hand against an older one.
  var R = D.results || null, RS = (R && R.steps) || {};
  if (R && R.run && state.runId !== R.run.id) {
    state.res = {}; state.note = {}; state.runId = R.run.id; state.view = 'results';
    Object.keys(R.values || {}).forEach(function (k) { delete state.data[k]; });
  }

  // Field registry: value suggestions and the short label shown when a value is still missing.
  var suggested = { BASE: M.appBase || '', ALIAS: M.alias || '' }, shortLabel = {}, fieldLabel = {}, placeholder = {};
  (D.data && D.data.groups || []).forEach(function (g) { (g.fields || []).forEach(function (f) {
    suggested[f.k] = f.value || ''; shortLabel[f.k] = f.short || f.label; fieldLabel[f.k] = f.label; placeholder[f.k] = f.placeholder || 'fill in';
  }); });
  if (R) Object.keys(R.values || {}).forEach(function (k) { suggested[k] = R.values[k]; });
  function val(k) { return ((k in state.data ? state.data[k] : suggested[k]) || '').trim(); }
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function base() { return (val('BASE') || '').replace(/\/+$/, ''); }
  function alias() { return val('ALIAS').replace(/^\/+|\/+$/g, ''); }
  function routePath(p) { return p.replace(/\[([A-Z][A-Z0-9_]*)\]/g, function (m, k) { return encodeURIComponent(val(k) || k); }); }
  function routeHref(p) { var a = alias(); return base() + '/' + (a ? a + '/' : '') + routePath(p); }
  // {KEY} becomes a value chip, {R:path} a link into the app.
  function fill(html) {
    return String(html || '').replace(/\{R:([^}]+)\}/g, function (m, p) {
      var a = alias();
      return '<a class="route" target="_blank" rel="noopener" href="' + esc(routeHref(p)) + '">/' + esc(a ? a + '/' : '') + esc(routePath(p)) + '</a>';
    }).replace(/\{([A-Z][A-Z0-9_]*)\}/g, function (m, k) {
      if (!(k in shortLabel)) return m;
      var v = val(k);
      return v ? '<span class="t">' + esc(v) + '</span>' : '<span class="t missing">' + esc(shortLabel[k]) + '?</span>';
    });
  }
  // Inside code blocks values go in raw, so the snippet can be pasted as is.
  function fillRaw(text) {
    return String(text || '').replace(/\{BASE\}/g, base()).replace(/\{ALIAS\}/g, alias()).replace(/\{([A-Z][A-Z0-9_]*)\}/g, function (m, k) {
      return k in shortLabel ? (val(k) || '<' + shortLabel[k] + '>') : m;
    });
  }
  var byId = {}; steps.forEach(function (s, i) { byId[s.id] = i; });
  var keyById = {}; key.forEach(function (r) { keyById[r.step] = r; });
  if (state.cur >= steps.length) state.cur = 0;
  var $ = function (id) { return document.getElementById(id); };
  var glyph = { pass: '✓', fail: '✕', skip: '–', blocked: '!', manual: '?' };
  // A viewer's own mark wins over the automated one; blocked and manual steps stay "to do" for the viewer.
  function autoRes(id) { return RS[id] ? RS[id].status : ''; }
  function effRes(id) { var a = autoRes(id); return state.res[id] || (a === 'pass' || a === 'fail' || a === 'skip' ? a : ''); }
  function dotRes(id) { return state.res[id] || autoRes(id); }

  // Tabs: only the ones with content.
  var tabs = [];
  if (R) tabs.push(['results', 'Results']);
  if (D.domain) tabs.push(['domain', 'Why it exists']);
  tabs.push(['run', 'Run']);
  if (D.data && D.data.groups && D.data.groups.length) tabs.push(['data', 'Test data']);
  if (key.length) tabs.push(['key', (D.key && D.key.tab) || 'Expected results']);
  if (D.how) tabs.push(['how', 'How it works']);
  if (R) tabs.push(['evidence', 'Evidence']);
  tabs.push(['sign', 'Sign-off']);
  $('tabs').innerHTML = tabs.map(function (t) { return '<button class="tab" role="tab" data-view="' + t[0] + '" aria-controls="v-' + t[0] + '">' + t[1] + '</button>'; }).join('');
  if (!tabs.some(function (t) { return t[0] === state.view; })) state.view = tabs[0][0];

  $('ticket-link').textContent = M.ticket; $('ticket-link').href = M.ticketUrl || '#';
  $('sub-title').textContent = M.title + (M.sliceLabel ? ' · ' + M.sliceLabel : '');
  document.title = M.ticket + ' Acceptance Run';

  // Why it exists
  if (D.domain) {
    function secs(list) { return (list || []).map(function (s) { return '<div class="sec">' + (s.h ? '<h2>' + esc(s.h) + '</h2>' : '') + fill(s.html) + '</div>'; }).join(''); }
    // Figures carry SVG, so their HTML goes in as is; live values come from <tspan data-t>.
    $('domain-figs').innerHTML = (D.domain.figures || []).map(function (f) { return '<div class="sec">' + (f.h ? '<h2>' + esc(f.h) + '</h2>' : '') + f.html + '</div>'; }).join('');
    $('domain-main').innerHTML = secs(D.domain.main);
    $('domain-side').innerHTML = secs(D.domain.side) + '<div class="sec"><p><a href="#" data-view-go="run">Start the run →</a></p></div>';
  }

  // Test data
  function renderData() {
    if (!D.data) return;
    $('data-lead').innerHTML = fill(D.data.lead || 'Prefilled where the environment already has suitable data. Fields marked on the left edge still need a value. Steps and expected results update as you type.');
    $('data-grid').innerHTML = D.data.groups.map(function (g) {
      var body;
      if (g.table) {
        body = '<div class="tbl-wrap"><table class="dt"><thead><tr><th></th>' + g.table.cols.map(function (c) { return '<th>' + esc(c) + '</th>'; }).join('') + '</tr></thead><tbody>' +
          g.table.rows.map(function (r) { return '<tr><th scope="row">' + esc(r.label) + '</th>' + r.cells.map(function (c) {
            if (c && c.k) return '<td><input class="in" id="d-' + c.k + '" data-k="' + c.k + '" aria-label="' + esc(fieldLabel[c.k]) + '" placeholder="' + esc(placeholder[c.k]) + '"></td>';
            return '<td class="na">' + fill(c && c.text != null ? c.text : (c || '')) + '</td>';
          }).join('') + '</tr>'; }).join('') + '</tbody></table></div>';
      } else {
        body = '<div class="tbl-wrap"><table class="dt"><tbody>' + g.fields.map(function (f) {
          return '<tr><th scope="row">' + esc(f.label) + '</th><td><input class="in" id="d-' + f.k + '" data-k="' + f.k + '" aria-label="' + esc(f.label) + '" placeholder="' + esc(placeholder[f.k]) + '"></td>' + (f.note ? '<td class="na">' + fill(f.note) + '</td>' : '') + '</tr>';
        }).join('') + '</tbody></table></div>';
      }
      return '<div class="sec"><h2>' + esc(g.title) + (g.note ? ' <span class="cap">' + esc(g.note) + '</span>' : '') + '</h2>' + body + (g.cap ? '<p class="cap">' + fill(g.cap) + '</p>' : '') + '</div>';
    }).join('');
    bindDataInputs();
  }
  function bindDataInputs() {
    document.querySelectorAll('#v-data input[data-k]').forEach(function (inp) {
      var k = inp.getAttribute('data-k'); inp.value = val(k);
      inp.oninput = function () { state.data[k] = inp.value; save(); refreshAll(true); };
    });
  }

  // Expected results
  function keyCell(c) {
    var v = c.v ? val(c.v) : '';
    var shown = c.v ? (v ? esc(v) : '<span class="t missing">' + esc(shortLabel[c.v] || c.v) + '?</span>') : fill(c.text || '');
    return '<span class="acc-cell">' + shown + '</span>' + (c.why ? ' <span class="src ' + (c.tag || 'def') + '">' + fill(c.why) + '</span>' : '');
  }
  function renderKey() {
    if (!key.length) return;
    var cols = D.key.cols || [];
    var h = (D.key.diagramHtml ? '<div class="sec">' + D.key.diagramHtml + '</div>' : '') +
      '<div class="sec"><h2>' + esc(D.key.title || 'Answer sheet') + '</h2>' + (D.key.lead ? '<p class="lead">' + fill(D.key.lead) + '</p>' : '') +
      '<div class="tbl-wrap key-wrap"><table class="dt key"><thead><tr><th>Step</th><th>' + esc(D.key.contextHead || 'Context') + '</th>' +
      cols.map(function (c) { return '<th>' + fill(c) + '</th>'; }).join('') + '</tr></thead><tbody>' +
      key.map(function (r) { return '<tr><td class="mono"><a href="#" data-go="' + r.step + '">' + r.step + '</a></td><td>' + fill(r.context) + '</td>' +
        r.items.map(function (c) { return '<td>' + keyCell(c) + '</td>'; }).join('') + '</tr>'; }).join('') + '</tbody></table></div></div>';
    $('key-inner').innerHTML = h;
    paintSvg($('key-inner'));
  }
  function expectTable(r) {
    var cols = D.key.cols || [];
    return '<table class="acct"><thead><tr><th>' + esc(D.key.itemHead || 'Item') + '</th><th>' + esc(D.key.valueHead || 'Expected') + '</th><th>Because</th></tr></thead><tbody>' +
      r.items.map(function (c, i) {
        var v = c.v ? val(c.v) : '';
        var big = c.v ? (v ? '<td class="big">' + esc(v) + '</td>' : '<td class="big missing">' + esc(shortLabel[c.v] || c.v) + '?</td>') : '<td class="big">' + fill(c.text || '') + '</td>';
        return '<tr><td>' + fill(cols[i] || '') + '</td>' + big + '<td>' + (c.why ? '<span class="src ' + (c.tag || 'def') + '">' + fill(c.why) + '</span>' : '') + '</td></tr>';
      }).join('') + '</tbody></table>';
  }

  // How it works
  function renderHow() {
    if (!D.how) return;
    $('how-diagrams').innerHTML = (D.how.sections || []).map(function (s) { return '<div class="sec">' + (s.h ? '<h2>' + esc(s.h) + '</h2>' : '') + s.html + '</div>'; }).join('');
    var devs = D.how.deviations || [];
    $('dev-list').innerHTML = devs.length ? devs.map(function (d) { return '<p class="callout"><b>' + esc(d.title) + '.</b> ' + fill(d.html) + '</p>'; }).join('') : '<p class="cap">No differences found between the ticket and the build.</p>';
    paintSvg($('how-diagrams'));
  }
  function paintSvg(root) {
    (root || document).querySelectorAll('svg [data-t]').forEach(function (el) {
      var k = el.getAttribute('data-t'), v = val(k);
      el.textContent = v || '????'; el.classList.toggle('missing', !v);
    });
  }

  // Run
  function renderRail() {
    var h = '', last = -1, opts = '';
    steps.forEach(function (s, i) {
      if (s.ph !== last) { h += '<h3>' + s.ph + ' · ' + esc(phases[s.ph].t) + '</h3>'; last = s.ph; }
      var r = dotRes(s.id), mine = state.res[s.id] && autoRes(s.id) && state.res[s.id] !== autoRes(s.id);
      h += '<button class="srow" data-i="' + i + '"' + (i === state.cur ? ' aria-current="step"' : '') + '><span class="dot ' + r + (mine ? ' you' : '') + '" aria-label="' + (r || 'to do') + (mine ? ', set by you' : '') + '">' + (glyph[r] || '') + '</span><span class="n">' + s.id + '</span><span class="ti">' + esc(s.t) + '</span></button>';
      opts += '<option value="' + i + '"' + (i === state.cur ? ' selected' : '') + '>' + s.id + ' ' + esc(s.t) + (r ? ' (' + r + ')' : '') + '</option>';
    });
    $('rail').innerHTML = h; $('pick').innerHTML = opts;
    var cur = $('rail').querySelector('[aria-current]'); if (cur) cur.scrollIntoView({ block: 'nearest' });
  }
  function whyHtml(s) {
    var failed = effRes(s.id) === 'fail';
    return '<section><h3>Phase ' + s.ph + '</h3><p class="phase-note">' + fill(phases[s.ph].d || '') + '</p></section>' +
      (s.b ? '<section><h3>Business meaning</h3><p>' + fill(s.b) + '</p></section>' : '') +
      (s.k ? '<details class="be"' + (failed ? ' open' : '') + '><summary>In the backend' + (failed ? '' : ' <span class="cap">for debugging</span>') + '</summary><p>' + fill(s.k) + '</p></details>' : '');
  }
  function fieldsHtml(s) {
    if (!s.fields || !s.fields.length) return '';
    return '<div class="inline-fields">' + s.fields.map(function (k) {
      return '<div class="field"><label for="f-' + k + '">' + esc(fieldLabel[k] || k) + '</label><input class="in" id="f-' + k + '" data-k="' + k + '" placeholder="' + esc(placeholder[k] || 'fill in') + '" value="' + esc(val(k)) + '"></div>';
    }).join('') + '</div>';
  }
  function codeHtml(s) {
    return (s.code || []).map(function (c, i) {
      return '<div class="code-blk"><div class="code-head"><span>' + esc(c.label || c.lang || 'snippet') + '</span><button class="copy-code" data-ci="' + i + '">Copy</button></div><pre><code>' + esc(fillRaw(c.text)) + '</code></pre></div>';
    }).join('');
  }
  function counts() { var c = { pass: 0, fail: 0, skip: 0, todo: 0 }; steps.forEach(function (s) { c[effRes(s.id) || 'todo']++; }); return c; }
  function renderStage() {
    var s = steps[state.cur];
    var h = '<div class="step-head"><span class="eyebrow">Phase ' + s.ph + ' · ' + esc(phases[s.ph].t) + '</span>' +
      '<h2><span class="num">' + s.id + '</span>' + esc(s.t) + '</h2>' +
      '<div class="acs">' + (s.ac || []).map(function (a) { return '<span class="ac">' + a + '</span>'; }).join('') + (s.optional ? '<span class="opt">optional</span>' : '') + '</div></div>';
    // After an automated run the outcome and its proof come first; the instructions follow as the reference.
    if (RS[s.id]) h += evHtml(s); else if (R) h += '<p class="ev-note">Not part of the automated run. Do it yourself and mark the result below.</p>';
    h += '<div class="blk"><span class="blk-l">' + (RS[s.id] ? 'What the step asks' : 'Do') + '</span><p>' + fill(s.d) + '</p></div>';
    h += fieldsHtml(s) + codeHtml(s);
    h += '<div class="blk expect"><span class="blk-l">Expect</span><p>' + fill(s.e) + '</p>' + (s.check && keyById[s.check] ? expectTable(keyById[s.check]) : '') +
      ((s.assume || []).length ? '<p class="assume"><b>Not confirmed in the app yet:</b> ' + s.assume.map(fill).join(' · ') + '</p>' : '') + '</div>';
    if (s.head) h += '<p class="callout"><b>Differs from the ticket.</b> ' + fill(s.head) + '</p>';
    if (s.known) h += '<p class="callout"><b>Expected to fail.</b> ' + fill(s.known) + '</p>';
    if (counts().todo === 0) h += '<p class="callout done-call">Every step has a result. <a href="#" data-view-go="sign">Open Sign-off</a> for the AC verdicts and the summary.</p>';
    h += '<div class="why-inline">' + whyHtml(s) + '</div>';
    var keep = $('stage').scrollTop, same = $('stage').getAttribute('data-step') === s.id;
    $('stage').innerHTML = h; $('stage').setAttribute('data-step', s.id); $('stage').scrollTop = same ? keep : 0;
    $('why').innerHTML = whyHtml(s) + '<p class="keys"><kbd>←</kbd> <kbd>→</kbd> move · <kbd>P</kbd> <kbd>F</kbd> <kbd>S</kbd> set result · Pass and Skip move on</p>';
    $('stage').querySelectorAll('input[data-k]').forEach(function (inp) {
      inp.addEventListener('input', function () {
        var k = inp.getAttribute('data-k'), pos = inp.selectionStart;
        state.data[k] = inp.value; save(); refreshAll(false);
        var again = $('f-' + k); if (again) { again.focus(); try { again.setSelectionRange(pos, pos); } catch (e) {} }
      });
    });
    $('stage').querySelectorAll('.copy-code').forEach(function (b) {
      b.addEventListener('click', function () {
        var txt = fillRaw(s.code[+b.getAttribute('data-ci')].text);
        try { navigator.clipboard.writeText(txt).then(function () { b.textContent = 'Copied'; }, function () { b.textContent = 'Select and copy'; }); } catch (e) { b.textContent = 'Select and copy'; }
      });
    });
    var r = effRes(s.id);
    document.querySelectorAll('.rbtn').forEach(function (b) { b.setAttribute('aria-pressed', String(b.getAttribute('data-r') === r)); });
    $('note-wrap').hidden = state.res[s.id] !== 'fail';
    $('note').value = state.note[s.id] || '';
    $('prev').disabled = state.cur === 0;
    $('next').disabled = state.cur === steps.length - 1;
    $('pos').textContent = (state.cur + 1) + ' of ' + steps.length;
    var nx = steps[state.cur + 1];
    $('upnext').innerHTML = nx ? '<span class="blk-l">Next · ' + nx.id + '</span><b>' + esc(nx.t) + '</b><p>' + fill(nx.d) + '</p>' : '<span class="blk-l">Last step</span><p>Open Sign-off for the AC verdicts and the summary.</p>';
    // A reviewer of an automated run reads results, not the next instruction; the space goes to the evidence.
    $('upnext').hidden = !!R;
  }
  function renderTally() {
    var c = counts(), n = steps.length;
    $('n-pass').textContent = c.pass; $('n-fail').textContent = c.fail; $('n-skip').textContent = c.skip; $('n-todo').textContent = c.todo;
    $('b-pass').style.width = (100 * c.pass / n) + '%'; $('b-fail').style.width = (100 * c.fail / n) + '%'; $('b-skip').style.width = (100 * c.skip / n) + '%';
  }

  // Sign-off
  // A step that checks several ACs can carry one verdict per AC (results acs), so a price check does not fail a posting AC.
  function acRes(id, ac) { var a = RS[id] && RS[id].acs; return state.res[id] || (ac && a && a[ac]) || effRes(id); }
  function acStatus(ids, ac) {
    var rs = ids.map(function (id) { return acRes(id, ac); });
    if (rs.indexOf('fail') >= 0) return 'fail';
    if (rs.every(function (r) { return r === 'pass' || r === 'skip'; }) && rs.indexOf('pass') >= 0) return 'pass';
    return 'todo';
  }
  function summaryText() {
    var c = counts(), lines = ['Acceptance run ' + M.ticket + ' (' + new Date().toISOString().slice(0, 10) + (alias() ? ', ' + alias() : '') + ')',
      'Steps: ' + c.pass + ' pass, ' + c.fail + ' fail, ' + c.skip + ' skip, ' + c.todo + ' to do'];
    if (R && R.run) {
      var t = evTotals();
      lines.push('Verified automatically by ' + (R.run.by || 'Claude') + ' on ' + String(R.run.finished || R.run.started || '').slice(0, 16).replace('T', ' ') +
        (R.run.env && R.run.env.app ? ' against ' + R.run.env.app.replace(/^https?:\/\//, '') + (R.run.env.tenant ? '/' + R.run.env.tenant : '') : '') + '.');
      lines.push('Evidence: ' + t.checks + ' checks, ' + t.shots + ' screenshots, ' + t.videos + ' videos, ' + t.http + ' API calls.');
      (R.run.env && R.run.env.builds || []).forEach(function (b) { lines.push('Build ' + b.label + ': ' + b.value); });
    }
    lines.push('');
    Object.keys(acMap).forEach(function (a) { var st = acStatus(acMap[a][1], a); lines.push(a + ': ' + (st === 'todo' ? 'not complete' : st.toUpperCase()) + ' - ' + acMap[a][0]); });
    var f = steps.filter(function (s) { return effRes(s.id) === 'fail'; });
    if (f.length) { lines.push('', 'Failures:'); f.forEach(function (s) { lines.push('- ' + s.id + ' ' + s.t + ': ' + failText(s)); }); }
    var open = steps.filter(function (s) { return !effRes(s.id) && RS[s.id]; });
    if (open.length) { lines.push('', 'Not verified automatically:'); open.forEach(function (s) { lines.push('- ' + s.id + ' ' + s.t + ' (' + RS[s.id].status + '): ' + plain(RS[s.id].summary || '')); }); }
    (M.knownIssues || []).forEach(function (k, i) { if (i === 0) lines.push(''); lines.push('Known issue: ' + k); });
    return lines.join('\n');
  }
  function renderSign() {
    var h = '<thead><tr><th>AC</th><th>Criterion</th><th>Steps</th><th>Status</th></tr></thead><tbody>';
    Object.keys(acMap).forEach(function (a) {
      var st = acStatus(acMap[a][1], a);
      h += '<tr><td class="mono">' + esc(a) + '</td><td>' + esc(acMap[a][0]) + '</td><td class="mono">' + acMap[a][1].map(function (id) {
        var r = acRes(id, a); return '<a href="#" data-go="' + id + '" title="' + (r || 'to do') + '">' + id + (r ? glyph[r] : '') + '</a>';
      }).join(' ') + '</td><td><span class="status ' + st + '">' + (st === 'todo' ? 'not complete' : st) + '</span></td></tr>';
    });
    $('ac-table').innerHTML = h + '</tbody>';
    var f = steps.filter(function (s) { return effRes(s.id) === 'fail'; });
    $('fail-list').innerHTML = f.length ? f.map(function (s) { return '<li><a href="#" data-go="' + s.id + '" class="mono">' + s.id + '</a> ' + esc(s.t) + ': ' + esc(failText(s)) + '</li>'; }).join('') : '<li class="cap">No failed steps.</li>';
    $('copy-src').value = summaryText();
  }

  function refreshHeader() {
    var a = alias(); $('app-link').href = base() + '/' + (a ? a + '/' : ''); $('app-link').textContent = base().replace(/^https?:\/\//, '') + (a ? '/' + a : '');
  }
  function refreshAll(fromDataTab) {
    refreshHeader(); renderKey(); renderHow(); renderStage(); renderSign();
    if (D.domain) paintSvg($('v-domain'));
    if (!fromDataTab) document.querySelectorAll('#v-data input[data-k]').forEach(function (inp) { if (document.activeElement !== inp) inp.value = val(inp.getAttribute('data-k')); });
  }
  function go(i) { state.cur = Math.max(0, Math.min(steps.length - 1, i)); save(); renderRail(); renderStage(); }
  function setResult(r) {
    var id = steps[state.cur].id;
    if (state.res[id] === r) delete state.res[id]; else state.res[id] = r;
    save(); renderTally(); renderSign(); renderResults();
    if (state.res[id] === 'fail') { renderRail(); renderStage(); $('note').focus(); return; }
    if (state.res[id]) { for (var i = state.cur + 1; i < steps.length; i++) if (!effRes(steps[i].id)) { go(i); return; } }
    renderRail(); renderStage();
  }
  function show(view) {
    state.view = view; save();
    document.querySelectorAll('.tab').forEach(function (t) { t.setAttribute('aria-selected', String(t.getAttribute('data-view') === view)); });
    document.querySelectorAll('.view').forEach(function (v) { v.hidden = v.id !== 'v-' + view; });
    if (view === 'sign') renderSign();
  }

  $('tabs').addEventListener('click', function (e) { var t = e.target.closest('.tab'); if (t) show(t.getAttribute('data-view')); });
  $('rail').addEventListener('click', function (e) { var b = e.target.closest('.srow'); if (b) go(+b.getAttribute('data-i')); });

function plain(html) {
  var d = document.createElement("div");
  d.innerHTML = fill(html);
  return d.textContent.replace(/\s+/g, " ").trim();
}
function fmtT(sec) {
  sec = Math.max(0, Math.round(sec || 0));
  return Math.floor(sec / 60) + ":" + ("0" + (sec % 60)).slice(-2);
}
function fmtMs(ms) {
  return ms == null
    ? ""
    : ms < 1000
    ? ms + " ms"
    : (ms / 1000).toFixed(1) + " s";
}
function timeOf(iso) {
  if (!iso) return "";
  var d = new Date(iso);
  return isNaN(d)
    ? String(iso)
    : d.toLocaleTimeString([], {
        hour: "2-digit",
        minute: "2-digit",
        second: "2-digit",
      });
}
function dateRange(a, b) {
  var d1 = new Date(a),
    d2 = b ? new Date(b) : null;
  if (isNaN(d1)) return String(a);
  var s =
    d1.toLocaleDateString([], {
      day: "numeric",
      month: "short",
      year: "numeric",
    }) +
    ", " +
    d1.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" });
  if (d2 && !isNaN(d2))
    s +=
      "–" +
      d2.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }) +
      " (" +
      Math.max(1, Math.round((d2 - d1) / 60000)) +
      " min)";
  return s;
}
function failText(s) {
  if (state.res[s.id] === "fail" || !RS[s.id])
    return state.note[s.id] || "no note";
  var r = RS[s.id],
    f = r.failure || {},
    bad = (r.checks || []).filter(function (c) {
      return !c.ok;
    })[0];
  var ex = f.expected != null ? f.expected : bad && bad.expected,
    ac = f.actual != null ? f.actual : bad && bad.actual;
  return (
    (r.cause ? "[" + (CAUSE_LABEL[r.cause] || r.cause) + "] " : "") +
    (ex != null ? 'expected "' + ex + '", got "' + ac + '". ' : "") +
    plain(r.summary || "")
  );
}
function evTotals() {
  var t = {
    checks: 0,
    checksOk: 0,
    shots: 0,
    http: 0,
    actions: 0,
    errors: 0,
    videos: 0,
    vsec: 0,
    by: { pass: 0, fail: 0, blocked: 0, skip: 0, manual: 0 },
    meth: { asserted: 0, observed: 0, manual: 0 },
  };
  ((R && R.videos) || []).forEach(function (v) {
    t.videos++;
    t.vsec += v.dur || 0;
  });
  Object.keys(RS).forEach(function (id) {
    var r = RS[id],
      c = r.checks || [];
    t.checks += c.length;
    t.checksOk += c.filter(function (x) {
      return x.ok;
    }).length;
    t.shots += (r.shots || []).length;
    t.http += (r.http || []).length;
    t.actions += (r.actions || []).length;
    t.errors += (r.console || []).filter(function (x) {
      return x.level === "error";
    }).length;
    if (r.status in t.by) t.by[r.status]++;
    if (r.method in t.meth) t.meth[r.method]++;
  });
  return t;
}

// Steps that fail for one reason (results issue) show as one item, not one red card per step.
function groupAtt(list) {
  var seen = {}, out = [];
  list.forEach(function (s) {
    var k = !state.res[s.id] && RS[s.id] && RS[s.id].issue;
    if (!k) return out.push(attItem(s));
    if (seen[k]) return seen[k].push(s);
    seen[k] = [s];
    out.push(k);
  });
  return out.map(function (x) {
    var g = seen[x];
    if (!g) return x;
    if (g.length === 1) return attItem(g[0]);
    // The group shows the cause most of its steps share.
    var n = {}, r0 = RS[g[0].id];
    g.forEach(function (s) { var c = RS[s.id].cause || ""; n[c] = (n[c] || 0) + 1; if (n[c] > (n[r0.cause || ""] || 0)) r0 = RS[s.id]; });
    return attItem(g[0])
      .replace(/<h3>[\s\S]*?<\/h3>/, "<h3>" + esc(x) + ' <span class="status ' + esc(r0.status) + '">' + g.length + " steps</span>" + causeChip(r0) + "</h3>")
      .replace(/<\/div><\/div>$/, '<div class="chips">' + g.map(function (s) { return chip(s.id); }).join("") + "</div></div></div>");
  });
}

      : fill(r.summary || f.diagnosis || "");
  return (
    '<div class="att-item' +
    (list.length ? "" : " nothumb") +
    '">' +
    (list.length ? thumb("step:" + s.id, list, 0, false) : "") +
    '<div><h3><a href="#" data-go="' +

    '" class="mono">' +

  );
}

// Viewer: screenshots zoom at the cursor and pan; recordings play one step at a time.
var LB = {
    list: [],
    i: 0,
    s: 1,
    x: 0,
    y: 0,
    fit: 1,
    mode: "img",
    vid: null,
    seg: null,
    keep: false,
    ret: null,

  ptr = {},
  pinch = null;
function showLb() {

function openShots(list, i) {

}
function loadShot() {
  var sh = LB.list[LB.i],
    img = $("lb-img"),
    s = steps[byId[sh.step]];
  $("lb-title").innerHTML =
    '<span class="mono">' + esc(sh.step) + "</span>" + esc(s ? s.t : "");

  $("lb-cap").innerHTML =

    (sh.kind
      ? '<span class="muted">' + esc(KIND_LABEL[sh.kind] || sh.kind) + "</span>"
      : "") +
    (sh.at ? '<span class="muted">' + esc(timeOf(sh.at)) + "</span>" : "") +
    (sh.url ? '<span class="muted">' + esc(sh.url) + "</span>" : "");
  $("lb-open").href = sh.src;
  $("lb-prev").hidden = $("lb-next").hidden = LB.list.length < 2;
  $("lb-chaps").innerHTML = "";
  $("lb-strip").innerHTML =
    LB.list.length > 1
      ? LB.list
          .map(function (x, j) {

              '<button data-li="' +

  if (cur) cur.scrollIntoView({ block: "nearest", inline: "nearest" });
  $("lb-hint").textContent =
    "Scroll or + − to zoom at the pointer · drag to move · double-click: fit or 100% · 0 fit · 1 actual size · ← → previous / next · Esc close";
  img.onload = fitImg;
  img.alt = sh.cap || "";
  img.src = sh.src;
  if (img.complete && img.naturalWidth) fitImg();
}
function box() {
  var r = $("lb-stage").getBoundingClientRect();
  return { w: r.width, h: r.height, l: r.left, t: r.top };
}
function applyT() {
  $("lb-img").style.transform =
    "translate(" + LB.x + "px," + LB.y + "px) scale(" + LB.s + ")";
  $("lb-zoom").textContent = Math.round(LB.s * 100) + "%";
}
function fitImg() {
  var img = $("lb-img"),
    b = box(),
    w = img.naturalWidth || 1,
    hh = img.naturalHeight || 1,
    pad = 20;
  var tall = hh / w > 2.2; // a full-page capture: fit its width and start at the top
  LB.fit = tall
    ? Math.min((b.w - 2 * pad) / w, 1)
    : Math.min((b.w - 2 * pad) / w, (b.h - 2 * pad) / hh, 1);
  LB.s = LB.fit;
  LB.x = (b.w - w * LB.s) / 2;
  LB.y = tall ? pad : Math.max(pad, (b.h - hh * LB.s) / 2);
  applyT();
}
function zoomAt(f, cx, cy) {
  var b = box();
  if (cx == null) {
    cx = b.w / 2;
    cy = b.h / 2;
  }
  var ns = Math.min(8, Math.max(0.05, LB.s * f));
  LB.x = cx - ((cx - LB.x) * ns) / LB.s;
  LB.y = cy - ((cy - LB.y) * ns) / LB.s;
  LB.s = ns;
  applyT();
}
function step(d) {
  if (LB.list.length < 2) return;
  LB.i = (LB.i + d + LB.list.length) % LB.list.length;
  loadShot();
}
var stage = $("lb-stage");
stage.addEventListener(
  "wheel",
  function (e) {
    if (LB.mode !== "img") return;
    e.preventDefault();
    var b = box();
    zoomAt(
      Math.exp(-e.deltaY * (e.deltaMode === 1 ? 0.05 : 0.0015)),
      e.clientX - b.l,
      e.clientY - b.t
    );
  },
  { passive: false }
);
stage.addEventListener("pointerdown", function (e) {
  if (LB.mode !== "img" || e.target.closest("button")) return;
  stage.setPointerCapture(e.pointerId);
  ptr[e.pointerId] = { x: e.clientX, y: e.clientY };
  stage.classList.add("drag");
  var ids = Object.keys(ptr);
  if (ids.length === 2) {
    var a = ptr[ids[0]],
      c = ptr[ids[1]];
    pinch = { d: Math.hypot(a.x - c.x, a.y - c.y), s: LB.s };
  }
});
stage.addEventListener("pointermove", function (e) {
  var p = ptr[e.pointerId];
  if (!p) return;
  var ids = Object.keys(ptr);
  if (ids.length === 2 && pinch) {
    ptr[e.pointerId] = { x: e.clientX, y: e.clientY };
    var a = ptr[ids[0]],
      c = ptr[ids[1]],
      b = box();
    zoomAt(
      (pinch.s * Math.hypot(a.x - c.x, a.y - c.y)) / pinch.d / LB.s,
      (a.x + c.x) / 2 - b.l,
      (a.y + c.y) / 2 - b.t
    );
    return;
  }
  LB.x += e.clientX - p.x;
  LB.y += e.clientY - p.y;
  ptr[e.pointerId] = { x: e.clientX, y: e.clientY };
  applyT();
});
function up(e) {
  delete ptr[e.pointerId];
  if (Object.keys(ptr).length < 2) pinch = null;
  if (!Object.keys(ptr).length) stage.classList.remove("drag");
}
stage.addEventListener("pointerup", up);
stage.addEventListener("pointercancel", up);
stage.addEventListener("dblclick", function (e) {
  if (LB.mode !== "img" || e.target.closest("button")) return;
  var b = box();
  if (Math.abs(LB.s - LB.fit) < 0.01)
    zoomAt(Math.max(1, LB.fit * 2) / LB.s, e.clientX - b.l, e.clientY - b.t);
  else fitImg();
});
window.addEventListener("resize", function () {
  if (!$("lb").hidden && LB.mode === "img" && Math.abs(LB.s - LB.fit) < 0.01)
    fitImg();
});

function lbKey(e) {
  var k = e.key;
  if (k === "Escape") {
    e.preventDefault();
    closeLb();
    return;
  }
  if (k === "Tab") {
    // keep focus inside the viewer
    var f = Array.prototype.filter.call(
      $("lb").querySelectorAll("button, a[href], video"),
      function (x) {
        return x.offsetParent !== null;
      }
    );
    if (!f.length) return;
    var i = f.indexOf(document.activeElement);
    if (e.shiftKey && i <= 0) {
      f[f.length - 1].focus();
      e.preventDefault();
    } else if (!e.shiftKey && i === f.length - 1) {
      f[0].focus();
      e.preventDefault();
    }
    return;
  }
  if (LB.mode !== "img") return;
  if (k === "ArrowRight") {
    step(1);
    e.preventDefault();
  } else if (k === "ArrowLeft") {
    step(-1);
    e.preventDefault();
  } else if (k === "+" || k === "=") zoomAt(1.25);
  else if (k === "-" || k === "_") zoomAt(0.8);
  else if (k === "0") fitImg();
  else if (k === "1") zoomAt(1 / LB.s);
}
